Lingvamyxa Lua 5.4 Integration Plan
==================================

Status

Working provisioning plan for vendoring Lua 5.4 into the Lingvamyxa
runtime and exposing it as an L5-style "lua" receiver. Companion to
include_python.txt; same verify / stage / generate / manifest pattern,
minus the runtime download -- Lua is a small C source tree built by the
project's own toolchain.

Goal

A tiny, embeddable scripting language for user glue: config, plugins,
rules, computed fields. LuaRocks packages are optional; most needs are
met by vendored .lua files.

Tier

Tier 2. ~250 KB of portable ANSI C. Drop it in third_party/lua, add a
static library target, link. Simpler than CPython.


1. Prerequisites (developer machine)
------------------------------------

Already in the project: CMake, Ninja, Git, a C toolchain. Nothing else.
uv or curl only to fetch the release tarball once for vendoring.


2. Vendor the source
--------------------

Source: https://www.lua.org/ftp/lua-5.4.7.tar.gz  (sha256 on the same
page; pin it). Commit third_party/lua.tar.gz, or the unpacked src/ tree.

The library is src/*.c minus src/lua.c (the REPL) and src/luac.c (the
compiler). Headers: lua.h, luaconf.h, lauxlib.h, lualib.h.


3. CMake integration
--------------------

    option(LINGVAMYXA_ENABLE_LUA "Build the lua receiver" OFF)

    if(LINGVAMYXA_ENABLE_LUA)
      file(GLOB LM_LUA_SRC ${CMAKE_BINARY_DIR}/libs/lua-src/src/*.c)
      list(REMOVE_ITEM LM_LUA_SRC .../lua.c .../luac.c)
      add_library(lingvamyxa_lua STATIC ${LM_LUA_SRC})
      target_include_directories(lingvamyxa_lua PUBLIC .../lua-src/src)
      target_compile_definitions(lingvamyxa_lua PRIVATE
        $<$<PLATFORM_ID:Linux>:LUA_USE_LINUX>
        $<$<PLATFORM_ID:Darwin>:LUA_USE_MACOSX>)
      target_compile_definitions(lingvamyxa_lua PUBLIC LINGVAMYXA_LUA=1)
    endif()

No install step: the code is linked into the runtime binary.


4. The C shim (luahost)  --  lm1/build/luahost.lm1.c
--------------------------------------------------

    lua_State *L = luaL_newstate();   luaL_openlibs(L);
    One lua_State per interpreter; create many, one per L5 thread if you
    want isolation with no shared lock.

    Call:  luaL_loadstring / lua_pcall, args pushed via lua_push*,
    results read with lua_to*.

    Arrays:  a contiguous LmArray maps to a Lua table of numbers (copy),
    or expose it as userdata + a metatable with __index / __len that
    reads the arena buffer in place (no copy). Lua 5.4 integers are
    64-bit; floats are doubles.

    Errors:  lua_pcall non-zero -> lua_tostring(L,-1) is the message;
    wrap with a traceback via lua_getglobal("debug"),"traceback" as the
    message handler; return an LMX failure record.

    No GIL. Each lua_State is single-threaded; never share one across
    L5 threads without your own lock.


5. ABI descriptors and the L5 receiver
--------------------------------------

    lm2/luahost_abi.lm2      predef + fn.external prototypes
    receiver "lua", selector "lua.eval" / "lua.call"

    lua.eval "return 2 + 2"
    lua.call "string.format" "%0.2f" x


6. Packages
-----------

Optional. LuaRocks needs a Lua interpreter to run and writes a tree you
can vendor:

    luarocks --tree build/libs/lua-modules install lpeg 1.1.0

Point package.path / package.cpath at build/libs/lua-modules in the
shim after luaL_openlibs. For most uses, vendoring .lua files directly
is simpler.


7. What the tool does not do
----------------------------

    write the body of lm_luahost_* (the table/userdata array bridge, the
    traceback handler);
    choose the LuaRocks set.


Appendix. Provisioning utility sketch  --  lm2/provision_lua.lm2
==============================================================

Shared helpers mirror lm2/vcpkgFetch.lm2 and are elided. Illustrative.

========================================================================
# lm2/provision_lua.lm2 -- vendor Lua 5.4, wire the lua receiver

include
constant
fn
prototype
unit

predef: "libc_abi.lm2"
predef: "lm2/own_abi.lm2"
include: "<stdio.h>" "<stdlib.h>" "<string.h>"

constant: LUA_VER "5.4.7"
constant: LUA_URL "https://www.lua.org/ftp/lua-5.4.7.tar.gz"
constant: LUA_SHA "REPLACE_ME"

fn: lm_lua_write_cmake () int
    FILE: f
    f: fopen("cmake/EmbeddedLua.cmake", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_lua.lm0\n"
    c.fprintf: f "option(LINGVAMYXA_ENABLE_LUA \"Build the lua receiver\" OFF)\n"
    c.fprintf: f "if(LINGVAMYXA_ENABLE_LUA)\n"
    c.fprintf: f "  set(LM_LUA_SRC_DIR \"${CMAKE_BINARY_DIR}/libs/lua-src/src\")\n"
    c.fprintf: f "  file(GLOB LM_LUA_SRC ${LM_LUA_SRC_DIR}/*.c)\n"
    c.fprintf: f "  list(REMOVE_ITEM LM_LUA_SRC ${LM_LUA_SRC_DIR}/lua.c ${LM_LUA_SRC_DIR}/luac.c)\n"
    c.fprintf: f "  add_library(lingvamyxa_lua STATIC ${LM_LUA_SRC})\n"
    c.fprintf: f "  target_include_directories(lingvamyxa_lua PUBLIC ${LM_LUA_SRC_DIR})\n"
    c.fprintf: f "  target_compile_definitions(lingvamyxa_lua PUBLIC LINGVAMYXA_LUA=1)\n"
    c.fprintf: f "endif()\n"
    fclose(f)
return: 0

fn: lm_lua_write_shim_stub () int
    FILE: f
    f: fopen("lm2/luahost_abi.lm2", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_lua.lm0\n"
    c.fprintf: f "predef: \"libc_abi.lm2\"\n\n"
    c.fprintf: f "external:\n"
    c.fprintf: f "    fn: lm_luahost_new () @: void\n"
    c.fprintf: f "    fn: lm_luahost_free (@: void state)\n"
    c.fprintf: f "    fn: lm_luahost_eval (@: void state; @: char code; @: void result_out) int\n"
    c.fprintf: f "end: external\n"
    fclose(f)
return: 0

external:
    fn: main (int: argc; @@: char argv) int
        c.array: []: char cmake 1024
        c.array: []: char archive 2048

        if: lm_enter_project_root(argv[0]) != 0
            return: 1
        strcpy(cmake, lm_env_or_default("LM_CMAKE", "cmake"))

        if: lm_join_path(archive, c.sizeof(archive), "third_party", "lua.tar.gz") != 0
            return: 1
        if: lm_fetch(LUA_URL, archive) != 0
            return: 1
        if: lm_verify_sha256(archive, LUA_SHA) != 0
            return: 1
        # lua-<ver>/ single top dir -> strip one component into lua-src
        if: lm_extract_strip1(cmake, archive, "build/libs/lua-src") != 0
            return: 1
        if: lm_lua_write_cmake() != 0
            return: 1
        if: lm_lua_write_shim_stub() != 0
            return: 1
        if: lm_write_manifest("build/libs/lua-src/PROVISION_MANIFEST.txt", "lua", LUA_VER, LUA_SHA) != 0
            return: 1

        c.printf: "provision_lua.lm0: done. build with -DLINGVAMYXA_ENABLE_LUA=ON\n"
    return: 0
end: external
========================================================================
