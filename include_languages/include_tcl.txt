Lingvamyxa Tcl Integration Plan
==============================

Status

Working provisioning plan for vendoring Tcl into the Lingvamyxa runtime
and exposing it as an L5-style "tcl" receiver. Companion to
include_python.txt; same verify / stage / generate / manifest pattern.
Tcl builds from a C source tree with autotools and stages a small script
library, so it sits between the pure-amalgamation Tier 2 entries and the
prebuilt Tier 1 ones.

Goal

Tcl is the language originally designed to be embedded in a C
application (Tool Command Language). Rock-solid multi-interpreter model,
excellent string and list handling, event loop, and a large legacy of
extensions. Good for user automation and command surfaces.

Tier

Tier 2. Vendor the source, `configure && make` with the project
toolchain into libtcl9.x (or libtcl8.6), stage the library/ script tree.


1. Prerequisites (developer machine)
------------------------------------

Already in the project: CMake, Ninja, Git, a C toolchain, plus a POSIX
shell + make for Tcl's unix/ configure (the project already relies on
sh/make for the bundled GMP/MPFR builds -- same requirement). Windows
uses the win/ makefile.


2. Vendor the source
--------------------

Source: core.tcl-lang.org / SourceForge:

    tcl<ver>-src.tar.gz        e.g. tcl9.0.0-src.tar.gz or tcl8.6.15-src.tar.gz

Pin the sha256. Commit third_party/tcl.tar.gz.


3. CMake integration
--------------------

Wrap the native build:

    option(LINGVAMYXA_ENABLE_TCL "Build the tcl receiver" OFF)

    if(LINGVAMYXA_ENABLE_TCL)
      add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/libs/tcl/lib/libtcl9.0.a
        COMMAND sh -c "cd ${CMAKE_BINARY_DIR}/libs/tcl-src/unix && \
          ./configure --prefix=${CMAKE_BINARY_DIR}/libs/tcl --disable-shared && \
          make && make install")
      add_custom_target(build_tcl DEPENDS .../libtcl9.0.a)
      add_library(lingvamyxa_tcl STATIC IMPORTED GLOBAL)
      set_target_properties(lingvamyxa_tcl PROPERTIES
        IMPORTED_LOCATION ${CMAKE_BINARY_DIR}/libs/tcl/lib/libtcl9.0.a)
      target_include_directories(lingvamyxa_tcl INTERFACE
        ${CMAKE_BINARY_DIR}/libs/tcl/include)
      add_dependencies(lingvamyxa_tcl build_tcl)
    endif()

`make install` also lays down build/libs/tcl/lib/tcl9.0/ -- the script
library. Stage it beside the binary and point TCL_LIBRARY at it.


4. The C shim (tclhost)  --  lm1/build/tclhost.lm1.c
--------------------------------------------------

    Tcl_FindExecutable(<exe_path>);           # helps Tcl locate its lib
    setenv("TCL_LIBRARY", <exe_dir>/../libs/tcl/lib/tcl9.0);
    Tcl_Interp *ip = Tcl_CreateInterp();
    Tcl_Init(ip);
    int rc = Tcl_Eval(ip, src);               # TCL_OK == 0
    const char *res = Tcl_GetStringResult(ip);
    Tcl_DeleteInterp(ip);

    Many interpreters; one per L5 thread (build with --enable-threads and
    use Tcl_CreateInterp per thread; interps are not shared across
    threads).

    Arrays:  Tcl_NewByteArrayObj (copy) for binary; for numeric data,
    Tcl_SetVar2Ex with a Tcl_Obj list (copy), or expose the arena buffer
    through a small command that reads/writes by index (no copy).

    Errors:  rc != TCL_OK -> Tcl_GetStringResult + the -errorinfo of the
    return options dict; format into an LMX failure record.


5. ABI descriptors and the L5 receiver
--------------------------------------

    lm2/tclhost_abi.lm2      predef + fn.external prototypes
    receiver "tcl", selector "tcl.eval"


6. Packages
-----------

Tcl extensions (Tcllib is pure Tcl) can be vendored under the staged
lib/ tree. Binary extensions (tDOM, tcllibc) would need their own
build; usually not required.


7. What the tool does not do
----------------------------

    write the body of lm_tclhost_* (the byte-array bridge, the
    errorInfo mapping);
    cross-compile Tcl (its configure is host-oriented; a cross build
    needs --host and a tcl for the build machine).


Appendix. Provisioning utility sketch  --  lm2/provision_tcl.lm2
==============================================================

Shared helpers mirror lm2/vcpkgFetch.lm2 and are elided. Illustrative.

========================================================================
# lm2/provision_tcl.lm2 -- vendor Tcl, build it, wire the tcl receiver

include
constant
fn
prototype
unit

predef: "libc_abi.lm2"
predef: "lm2/own_abi.lm2"
include: "<stdio.h>" "<stdlib.h>" "<string.h>"

constant: TCL_VER "9.0.0"
constant: TCL_URL "https://downloads.sourceforge.net/tcl/tcl9.0.0-src.tar.gz"
constant: TCL_SHA "REPLACE_ME"

fn: lm_tcl_write_cmake () int
    FILE: f
    f: fopen("cmake/EmbeddedTcl.cmake", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_tcl.lm0\n"
    c.fprintf: f "option(LINGVAMYXA_ENABLE_TCL \"Build the tcl receiver\" OFF)\n"
    c.fprintf: f "if(LINGVAMYXA_ENABLE_TCL)\n"
    c.fprintf: f "  set(S \"${CMAKE_BINARY_DIR}/libs/tcl-src\")\n"
    c.fprintf: f "  set(P \"${CMAKE_BINARY_DIR}/libs/tcl\")\n"
    c.fprintf: f "  add_custom_command(OUTPUT ${P}/lib/libtcl9.0.a\n"
    c.fprintf: f "    COMMAND sh -c \"cd ${S}/unix && ./configure --prefix=${P} --disable-shared && make && make install\")\n"
    c.fprintf: f "  add_custom_target(build_tcl DEPENDS ${P}/lib/libtcl9.0.a)\n"
    c.fprintf: f "  add_library(lingvamyxa_tcl STATIC IMPORTED GLOBAL)\n"
    c.fprintf: f "  set_target_properties(lingvamyxa_tcl PROPERTIES IMPORTED_LOCATION ${P}/lib/libtcl9.0.a)\n"
    c.fprintf: f "  target_include_directories(lingvamyxa_tcl INTERFACE ${P}/include)\n"
    c.fprintf: f "  add_dependencies(lingvamyxa_tcl build_tcl)\n"
    c.fprintf: f "endif()\n"
    fclose(f)
return: 0

fn: lm_tcl_write_shim_stub () int
    FILE: f
    f: fopen("lm2/tclhost_abi.lm2", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_tcl.lm0\n"
    c.fprintf: f "predef: \"libc_abi.lm2\"\n\n"
    c.fprintf: f "external:\n"
    c.fprintf: f "    fn: lm_tclhost_new () @: void\n"
    c.fprintf: f "    fn: lm_tclhost_free (@: void ip)\n"
    c.fprintf: f "    fn: lm_tclhost_eval (@: void ip; @: char code; @: void result_out) int\n"
    c.fprintf: f "end: external\n"
    fclose(f)
return: 0

external:
    fn: main (int: argc; @@: char argv) int
        c.array: []: char cmake 1024
        c.array: []: char archive 2048

        if: lm_enter_project_root(argv[0]) != 0
            return: 1
        strcpy(cmake, lm_env_or_default("LM_CMAKE", "cmake"))

        if: lm_join_path(archive, c.sizeof(archive), "third_party", "tcl.tar.gz") != 0
            return: 1
        if: lm_fetch(TCL_URL, archive) != 0
            return: 1
        if: lm_verify_sha256(archive, TCL_SHA) != 0
            return: 1
        if: lm_extract_strip1(cmake, archive, "build/libs/tcl-src") != 0
            return: 1
        if: lm_tcl_write_cmake() != 0
            return: 1
        if: lm_tcl_write_shim_stub() != 0
            return: 1
        if: lm_write_manifest("build/libs/tcl-src/PROVISION_MANIFEST.txt", "tcl", TCL_VER, TCL_SHA) != 0
            return: 1

        c.printf: "provision_tcl.lm0: done. build with -DLINGVAMYXA_ENABLE_TCL=ON\n"
    return: 0
end: external
========================================================================
