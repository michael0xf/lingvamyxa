Lingvamyxa mruby Integration Plan
================================

Status

Working provisioning plan for vendoring mruby into the Lingvamyxa
runtime and exposing it as an Message Thread "mruby" receiver. Companion to
include_python.txt; same verify / stage / generate / manifest pattern,
no runtime download, with one bootstrap wrinkle (the build runs on a
host Ruby).

Goal

Ruby syntax and semantics for embedding: small footprint, many
independent VM states, a build-time "gembox" that compiles selected
mgems straight into libmruby.a. For user automation with Ruby ergonomics
where full CRuby (and its C-extension gems) is not needed.

Tier

Tier 2. Self-contained C, but its build system (`rake` / bundled
`minirake`) needs a Ruby interpreter to run. mruby ships one; on a
machine with no Ruby at all, a tiny host ruby or a prebuilt mruby is
required for the first build.


1. Prerequisites (developer machine)
------------------------------------

Already in the project: CMake, Ninja, Git, a C toolchain.

Add:

    a host Ruby        to run mruby's build (any 2.x/3.x; mruby also
                       bundles minirake, but rake needs ruby to execute)
    uv or curl         fetch the release once


2. Vendor the source
--------------------

Source: mruby.org / github.com/mruby/mruby releases:

    mruby-<ver>.tar.gz         e.g. mruby-3.3.0.tar.gz

Pin the sha256. Commit third_party/mruby.tar.gz.

Add a build config that selects the gembox:

    # third_party/mruby-build/lingva.rb
    MRuby::Build.new do |conf|
      conf.toolchain
      conf.gembox 'default'
      conf.gem :core => 'mruby-eval'
      conf.gem :mgem => 'mruby-json'
      conf.gem :mgem => 'mruby-regexp-pcre'   # links against bundled pcre2
    end


3. CMake integration
--------------------

Wrap the native build:

    option(LINGVAMYXA_ENABLE_MRUBY "Build the mruby receiver" OFF)

    if(LINGVAMYXA_ENABLE_MRUBY)
      add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/libs/mruby/build/host/lib/libmruby.a
        COMMAND ${CMAKE_COMMAND} -E env
          MRUBY_CONFIG=${CMAKE_SOURCE_DIR}/third_party/mruby-build/lingva.rb
          CC=${CMAKE_C_COMPILER}
          rake
        WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/libs/mruby)
      add_custom_target(build_mruby DEPENDS .../libmruby.a)
      add_library(lingvamyxa_mruby STATIC IMPORTED GLOBAL)
      set_target_properties(lingvamyxa_mruby PROPERTIES
        IMPORTED_LOCATION ${CMAKE_BINARY_DIR}/libs/mruby/build/host/lib/libmruby.a)
      target_include_directories(lingvamyxa_mruby INTERFACE
        ${CMAKE_BINARY_DIR}/libs/mruby/include)
      add_dependencies(lingvamyxa_mruby build_mruby)
    endif()


4. The C shim (mrubyhost)  --  lm1/build/mrubyhost.lm1.c
------------------------------------------------------

    mrb_state *mrb = mrb_open();
    mrb_value v = mrb_load_string(mrb, src);
    if (mrb->exc) { format mrb_inspect(mrb, mrb_obj_value(mrb->exc)) ->
        LMX failure record; mrb->exc = 0; }
    mrb_close(mrb);

    Many states; one mrb_state per Message Thread (states are not shared
    across threads).

    Arrays:  mrb_str_new (copy) for binary; for numeric data, an
    mrb_ary_new (copy) or a small C-defined class wrapping the arena
    pointer with [] / []= / size (no copy). mruby floats are doubles;
    integers are 63-bit or boxed.

    No GIL, but a state is single-threaded.


5. ABI descriptors and the Message Thread receiver
--------------------------------------

    lm2/mrubyhost_abi.lm2    predef + fn.external prototypes
    receiver "mruby", selector "mruby.eval" / "mruby.call"


6. Packages
-----------

mgems are selected in the build config (section 2) and compiled into
libmruby.a. There is no runtime gem install. Adding a gem = edit
lingva.rb and rebuild.


7. What the tool does not do
----------------------------

    write the body of lm_mrubyhost_* (the array class, the exception
    inspect mapping);
    choose the gembox;
    provide a host Ruby (must be on PATH for the build).


Appendix. Provisioning utility sketch  --  lm2/provision_mruby.lm2
==============================================================

Shared helpers mirror lm2/vcpkgFetch.lm2 and are elided. Illustrative.

========================================================================
# lm2/provision_mruby.lm2 -- vendor mruby, build it, wire the mruby receiver

include
constant
fn
prototype
unit

predef: "libc_abi.lm2"
predef: "lm2/own_abi.lm2"
include: "<stdio.h>" "<stdlib.h>" "<string.h>"

constant: MRUBY_VER "3.3.0"
constant: MRUBY_URL "https://github.com/mruby/mruby/archive/refs/tags/3.3.0.tar.gz"
constant: MRUBY_SHA "REPLACE_ME"

fn: lm_mruby_write_cmake () int
    FILE: f
    f: fopen("cmake/EmbeddedMruby.cmake", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_mruby.lm0\n"
    c.fprintf: f "option(LINGVAMYXA_ENABLE_MRUBY \"Build the mruby receiver\" OFF)\n"
    c.fprintf: f "if(LINGVAMYXA_ENABLE_MRUBY)\n"
    c.fprintf: f "  set(M \"${CMAKE_BINARY_DIR}/libs/mruby\")\n"
    c.fprintf: f "  add_custom_command(OUTPUT ${M}/build/host/lib/libmruby.a\n"
    c.fprintf: f "    COMMAND ${CMAKE_COMMAND} -E env MRUBY_CONFIG=${CMAKE_SOURCE_DIR}/third_party/mruby-build/lingva.rb CC=${CMAKE_C_COMPILER} rake\n"
    c.fprintf: f "    WORKING_DIRECTORY ${M})\n"
    c.fprintf: f "  add_custom_target(build_mruby DEPENDS ${M}/build/host/lib/libmruby.a)\n"
    c.fprintf: f "  add_library(lingvamyxa_mruby STATIC IMPORTED GLOBAL)\n"
    c.fprintf: f "  set_target_properties(lingvamyxa_mruby PROPERTIES IMPORTED_LOCATION ${M}/build/host/lib/libmruby.a)\n"
    c.fprintf: f "  target_include_directories(lingvamyxa_mruby INTERFACE ${M}/include)\n"
    c.fprintf: f "  add_dependencies(lingvamyxa_mruby build_mruby)\n"
    c.fprintf: f "endif()\n"
    fclose(f)
return: 0

fn: lm_mruby_write_shim_stub () int
    FILE: f
    f: fopen("lm2/mrubyhost_abi.lm2", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_mruby.lm0\n"
    c.fprintf: f "predef: \"libc_abi.lm2\"\n\n"
    c.fprintf: f "external:\n"
    c.fprintf: f "    fn: lm_mrubyhost_new () @: void\n"
    c.fprintf: f "    fn: lm_mrubyhost_free (@: void mrb)\n"
    c.fprintf: f "    fn: lm_mrubyhost_eval (@: void mrb; @: char code; @: void result_out) int\n"
    c.fprintf: f "end: external\n"
    fclose(f)
return: 0

external:
    fn: main (int: argc; @@: char argv) int
        c.array: []: char cmake 1024
        c.array: []: char archive 2048

        if: lm_enter_project_root(argv[0]) != 0
            return: 1
        strcpy(cmake, lm_env_or_default("LM_CMAKE", "cmake"))

        if: lm_join_path(archive, c.sizeof(archive), "third_party", "mruby.tar.gz") != 0
            return: 1
        if: lm_fetch(MRUBY_URL, archive) != 0
            return: 1
        if: lm_verify_sha256(archive, MRUBY_SHA) != 0
            return: 1
        if: lm_extract_strip1(cmake, archive, "build/libs/mruby") != 0
            return: 1
        if: lm_mruby_write_cmake() != 0
            return: 1
        if: lm_mruby_write_shim_stub() != 0
            return: 1
        if: lm_write_manifest("build/libs/mruby/PROVISION_MANIFEST.txt", "mruby", MRUBY_VER, MRUBY_SHA) != 0
            return: 1

        c.printf: "provision_mruby.lm0: done (needs a host ruby to build). -DLINGVAMYXA_ENABLE_MRUBY=ON\n"
    return: 0
end: external
========================================================================
