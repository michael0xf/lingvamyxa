Lingvamyxa MuJS / JerryScript Integration Plan
=============================================

Status

Working provisioning plan for vendoring a kilobyte-class JavaScript
engine -- MuJS or JerryScript -- into the Lingvamyxa runtime and
exposing it as an Message Thread "js" receiver. Companion to
include_python.txt; same verify / stage / generate / manifest pattern,
no runtime download.

Goal

The smallest possible JS interpreter for constrained targets or when the
Lingvamyxa binary itself must stay tiny. Two options in one plan:

    MuJS         ES5, ~200 KB, from the MuPDF project, plain make;
    JerryScript  ES 5.1 with a subset of later features, CMake, built for
                 the 64 KB RAM class, WeakRef/typed arrays available.

For ES2023 use include_quickjs.txt instead.

Tier

Tier 2. A handful of C files, project toolchain, static lib.


1. Prerequisites (developer machine)
------------------------------------

Already in the project: CMake, Ninja, Git, a C toolchain. uv or curl to
fetch the release once.


2. Vendor the source
--------------------

    MuJS         from mujs.com / the git mirror; ~10 .c files, or the
                 generated one.c amalgamation. Headers: mujs.h.
    JerryScript  from the jerryscript-project releases; CMake tree, port
                 layer under jerry-port/. Headers: jerryscript.h.

Pin the release sha256. Commit third_party/mujs/ (or third_party/jerry/).


3. CMake integration
--------------------

    option(LINGVAMYXA_ENABLE_JS "Build the js receiver" OFF)
    set(LINGVAMYXA_JS_IMPL "mujs" CACHE STRING "mujs | jerry")

    if(LINGVAMYXA_ENABLE_JS AND LINGVAMYXA_JS_IMPL STREQUAL "mujs")
      add_library(lingvamyxa_js STATIC
        ${CMAKE_SOURCE_DIR}/third_party/mujs/one.c)
      target_include_directories(lingvamyxa_js PUBLIC
        ${CMAKE_SOURCE_DIR}/third_party/mujs)
    endif()
    # JerryScript: add_subdirectory(third_party/jerry) and link
    # jerry-core (its CMake exports the target).


4. The C shim (jshost)  --  lm1/build/jshost.lm1.c
------------------------------------------------

MuJS:

    js_State *J = js_newstate(NULL, NULL, JS_STRICT);
    js_dostring(J, src);                    # 0 = ok
    js_freestate(J);
    Errors: install a try/catch via js_try or a panic handler; the error
    object's stack/message becomes an LMX failure record.
    Arrays: MuJS has no typed arrays; pass numeric data as a JS array
    (copy) or as an opaque userdata with accessor functions.

JerryScript:

    jerry_init(JERRY_INIT_EMPTY);
    jerry_value_t r = jerry_eval((const jerry_char_t*)src, len, JERRY_PARSE_NO_OPTS);
    jerry_value_is_exception(r) -> jerry_exception_value -> to string;
    jerry_cleanup();
    Arrays: jerry_arraybuffer_external / jerry_typedarray for a
    no-copy view over the arena buffer.

No GIL in either; one state per Message Thread.


5. ABI descriptors and the Message Thread receiver
--------------------------------------

    lm2/jshost_abi.lm2       predef + fn.external prototypes
    receiver "js", selector "js.eval" / "js.call"
    (share the name with include_quickjs.txt; the impl is a build switch)


6. Packages
-----------

None. Vendor pure-JS (ES5 for MuJS) sources and load in order.


7. What the tool does not do
----------------------------

    write the body of lm_jshost_* for the chosen impl;
    provide typed-array bridging for MuJS (it has none -- copy).


Appendix. Provisioning utility sketch  --  lm2/provision_mujs.lm2
==============================================================

Shared helpers mirror lm2/vcpkgFetch.lm2 and are elided. Illustrative;
shows the MuJS path.

========================================================================
# lm2/provision_mujs.lm2 -- vendor MuJS, wire the js receiver (impl=mujs)

include
constant
fn
prototype
unit

predef: "libc_abi.lm2"
predef: "lm2/own_abi.lm2"
include: "<stdio.h>" "<stdlib.h>" "<string.h>"

constant: MUJS_VER "1.3.5"
constant: MUJS_URL "https://mujs.com/downloads/mujs-1.3.5.tar.gz"
constant: MUJS_SHA "REPLACE_ME"

fn: lm_mujs_write_cmake () int
    FILE: f
    f: fopen("cmake/EmbeddedJs.cmake", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_mujs.lm0\n"
    c.fprintf: f "option(LINGVAMYXA_ENABLE_JS \"Build the js receiver\" OFF)\n"
    c.fprintf: f "if(LINGVAMYXA_ENABLE_JS)\n"
    c.fprintf: f "  set(D \"${CMAKE_SOURCE_DIR}/third_party/mujs\")\n"
    c.fprintf: f "  add_library(lingvamyxa_js STATIC ${D}/one.c)\n"
    c.fprintf: f "  target_include_directories(lingvamyxa_js PUBLIC ${D})\n"
    c.fprintf: f "  target_compile_definitions(lingvamyxa_js PUBLIC LINGVAMYXA_JS=1 LINGVAMYXA_JS_MUJS=1)\n"
    c.fprintf: f "endif()\n"
    fclose(f)
return: 0

fn: lm_mujs_write_shim_stub () int
    FILE: f
    f: fopen("lm2/jshost_abi.lm2", "wb")
    if: f = 0
        return: 1
    c.fprintf: f "# generated by provision_mujs.lm0\n"
    c.fprintf: f "predef: \"libc_abi.lm2\"\n\n"
    c.fprintf: f "external:\n"
    c.fprintf: f "    fn: lm_jshost_new () @: void\n"
    c.fprintf: f "    fn: lm_jshost_free (@: void state)\n"
    c.fprintf: f "    fn: lm_jshost_eval (@: void state; @: char code; @: void result_out) int\n"
    c.fprintf: f "end: external\n"
    fclose(f)
return: 0

external:
    fn: main (int: argc; @@: char argv) int
        c.array: []: char cmake 1024
        c.array: []: char archive 2048

        if: lm_enter_project_root(argv[0]) != 0
            return: 1
        strcpy(cmake, lm_env_or_default("LM_CMAKE", "cmake"))

        if: lm_join_path(archive, c.sizeof(archive), "third_party", "mujs.tar.gz") != 0
            return: 1
        if: lm_fetch(MUJS_URL, archive) != 0
            return: 1
        if: lm_verify_sha256(archive, MUJS_SHA) != 0
            return: 1
        if: lm_extract_strip1(cmake, archive, "third_party/mujs") != 0
            return: 1
        # generate one.c if the tarball lacks it: cat the *.c into one.c
        if: lm_mujs_write_cmake() != 0
            return: 1
        if: lm_mujs_write_shim_stub() != 0
            return: 1
        if: lm_write_manifest("third_party/mujs/PROVISION_MANIFEST.txt", "mujs", MUJS_VER, MUJS_SHA) != 0
            return: 1

        c.printf: "provision_mujs.lm0: done. build with -DLINGVAMYXA_ENABLE_JS=ON\n"
    return: 0
end: external
========================================================================
